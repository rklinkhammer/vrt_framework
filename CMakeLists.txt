cmake_minimum_required(VERSION 3.25)
project(vita49 VERSION 0.1.0 LANGUAGES CXX)

include(CTest)
include(CheckCXXSourceCompiles)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/cmake/cxx23_probe.cpp" VITA_CXX23_PROBE)
check_cxx_source_compiles("${VITA_CXX23_PROBE}" VITA_HAS_REQUIRED_CXX23)
if(NOT VITA_HAS_REQUIRED_CXX23)
  message(FATAL_ERROR "VITA requires C++23 with std::expected, std::span, std::bit_cast, std::endian, std::byteswap and concepts. Select a compiler AND standard library implementing these facilities. See CMakeFiles/CMakeConfigureLog.yaml.")
endif()

add_library(vita_core INTERFACE)
add_library(vita::core ALIAS vita_core)
target_include_directories(vita_core INTERFACE
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)
target_compile_features(vita_core INTERFACE cxx_std_23)
# Default RTTI-free core; no layout-changing public configuration macros.
if(MSVC)
  target_compile_options(vita_core INTERFACE /GR-)
else()
  target_compile_options(vita_core INTERFACE -fno-rtti)
endif()

set(VITA_SANITIZER "none" CACHE STRING "none, address-undefined, or thread")
set_property(CACHE VITA_SANITIZER PROPERTY STRINGS none address-undefined thread)
if(NOT VITA_SANITIZER STREQUAL "none")
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    message(FATAL_ERROR "Sanitizer presets require Clang or GCC")
  endif()
  if(VITA_SANITIZER STREQUAL "address-undefined")
    set(VITA_SANITIZER_FLAG -fsanitize=address,undefined)
  elseif(VITA_SANITIZER STREQUAL "thread")
    set(VITA_SANITIZER_FLAG -fsanitize=thread)
  else()
    message(FATAL_ERROR "Unknown VITA_SANITIZER=${VITA_SANITIZER}")
  endif()
  add_compile_options(${VITA_SANITIZER_FLAG} -fno-omit-frame-pointer)
  add_link_options(${VITA_SANITIZER_FLAG})
endif()

function(vita_add_test name)
  add_executable(${name} ${ARGN})
  target_link_libraries(${name} PRIVATE vita::core)
  if(MSVC)
    target_compile_options(${name} PRIVATE /W4 /EHs-c- /UNDEBUG)
  else()
    target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -fno-exceptions -UNDEBUG)
  endif()
  add_test(NAME ${name} COMMAND ${name})
endfunction()
