function(vita_scan_target target)
  add_executable(${target} main.cpp)
  target_link_libraries(${target} PRIVATE vita::core)
  if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(${target} PRIVATE -fno-exceptions -fno-rtti)
  endif()
endfunction()
vita_scan_target(vita_frequency_scan_combined)
if(VITA_BUILD_POSIX_UDP)
  vita_scan_target(vita_frequency_scan_controller)
  vita_scan_target(vita_frequency_scan_controllee)
  target_compile_definitions(vita_frequency_scan_controller PRIVATE VITA_SCAN_UDP=1)
  target_compile_definitions(vita_frequency_scan_controllee PRIVATE VITA_SCAN_UDP=1 VITA_SCAN_CONTROLLEE=1)
  target_link_libraries(vita_frequency_scan_controller PRIVATE vita::posix_udp)
  target_link_libraries(vita_frequency_scan_controllee PRIVATE vita::posix_udp)
endif()
if(BUILD_TESTING)
  add_test(NAME p16_example_combined COMMAND vita_frequency_scan_combined --deterministic --dwell-ms 1)
  add_test(NAME p16_example_wallclock COMMAND vita_frequency_scan_combined --start-hz 100025000 --stop-hz 100025000 --dwell-ms 1)
  set_tests_properties(p16_example_combined p16_example_wallclock PROPERTIES TIMEOUT 15 LABELS "P16;examples;integration")
  if(VITA_BUILD_POSIX_UDP)

    add_test(NAME p16_example_udp COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/run_pair.py --controller $<TARGET_FILE:vita_frequency_scan_controller> --controllee $<TARGET_FILE:vita_frequency_scan_controllee>)
    add_test(NAME p16_example_timeout COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/run_pair.py --controller $<TARGET_FILE:vita_frequency_scan_controller> --timeout-only)
    set_tests_properties(p16_example_udp p16_example_timeout PROPERTIES TIMEOUT 20 RESOURCE_LOCK p16_process_ports LABELS "P16;examples;integration;udp")
  endif()
endif()
